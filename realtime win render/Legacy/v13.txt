#include <stdio.h>
#include <stdlib.h>
#include <math.h>

#define width 1280
#define height 720
//range of canvas coordinates in code: 0 ~ width - 1, 0 ~ height - 1

//LV = location vector
//RV = rotation vector, could be unit vector or not a unit vector
//all vectors are stored in double type
//all angles are done in radians
//for pixel coordinates, higher for x = right, higher for y = down

//rgb are stored in char because it's 1 byte so it's efficient for storing numbers within 0 ~ 255
//needs to be unsigned or it gives minus rgb values
typedef struct {
	unsigned char r;
	unsigned char b;
	unsigned char g;
} RGB;

typedef struct {
	double u;
	double v;
	double w;
} UV;

typedef struct {
	double x;
	double y;
	double z;
} vector;

//data inside struct 'face' are all indeces, NOT VALUES
//all are pointers pointing to an array of indices
//negative indeces are interpreted into a positive index in reading stage
//indices start at 1, 0 means unused.
typedef struct {
	unsigned int vertCount;
	unsigned int* point;
	unsigned int* texture;
	unsigned int* normal;
} face;

typedef struct {
	char* name;
	face* mesh;
} object;

//////////////////NEEDS TO BECOME DYNAMIC!!

void exitSafety(FILE* placeholdA, FILE* placeholdB, void* placeholdC) {
	fclose(placeholdA);
	fclose(placeholdB);
	free(placeholdC);
	return;
}

vector vectorSubtract(vector vectA, vector vectB) {
	vector calcResult;
	calcResult.x = vectA.x - vectB.x;
	calcResult.y = vectA.y - vectB.y;
	calcResult.z = vectA.z - vectB.z;
	return calcResult;
}

vector vectorScale(double scalar, vector vect) {
	vector calcResult;
	calcResult.x = scalar * vect.x;
	calcResult.y = scalar * vect.y;
	calcResult.z = scalar * vect.z;
	return calcResult;
}

void drawLine(RGB (*drawTo)[height], int aX, int aY, int bX, int bY, RGB lineCol) {
	int cursor, advance;
	//cursor glides through the chose axis, couple is the other axis.
	float couple, slope;
	//slope is RELATIVE TO MAIN AXIS! it's not always dx/dy
	if (fabsf((float)(aY - bY) / (float)(aX - bX)) > 1) {
		//cursor advances in Y direction
		cursor = aY;
		if (aY == bY)
			advance = 0;
		else
			advance = (bY - aY) / abs(bY - aY);
		couple = (float)aX;
		slope = ((float)(aX - bX) / (float)(aY - bY)) * advance;
		while (cursor != bY) {
			drawTo[(int)couple][cursor].r = lineCol.r;
			drawTo[(int)couple][cursor].g = lineCol.g;
			drawTo[(int)couple][cursor].b = lineCol.b;
			cursor += advance;
			couple += slope;
		}
	}
	else {
		//cursor advances in X direction
		cursor = aX;
		if (aX == bX)
			advance = 0;
		else
			advance = (bX - aX) / abs(bX - aX);
		couple = (float)aY;
		slope = ((float)(aY - bY) / (float)(aX - bX)) * advance;
		while (cursor != bX) {
			drawTo[cursor][(int)couple].r = lineCol.r;
			drawTo[cursor][(int)couple].g = lineCol.g;
			drawTo[cursor][(int)couple].b = lineCol.b;
			cursor += advance;
			couple += slope;
		}
	}
	return;
}

////////////////////////////////PLACEHOLDER
void drawCircle(RGB (*drawTo)[height], int posX, int posY, double radius, RGB fillCol) {
	drawTo[posX][posY] = fillCol;
	drawTo[posX + 1][posY] = fillCol;
	drawTo[posX - 1][posY] = fillCol;
	drawTo[posX][posY + 1] = fillCol;
	drawTo[posX][posY - 1] = fillCol;
	return;
}

//FOV is measured on the X axis
//size of projection plane: X = cameraScale, Y = (resolutionY / resolutionX) * cameraScale
void main() {
	int widthHalf = width / 2, heightHalf = height / 2, inpBufferCursor, indexBufferCursor;
	double FOV = 120, cameraScale = 1, perspDist;
	vector perspLV = { .x = 0, .y = 0, .z = 0 };
	vector vectorBuffer;
	int vertexCount = 0, vertNormalCount = 0, vertTextureCount = 0, faceCount = 0, shadeSmooth = 0, vertAttribute[3], indexType;
	char inputBuffer[256], indexBuffer[32];
	vector cameraLV = { .x = 0, .y = 0, .z = 0 }, cameraRV = { .x = 1, .y = 0, .z = 0 };
	vector(*meshVertex) = (vector*)malloc(0);
	vector(*meshVertNormal) = (vector*)malloc(0);
	UV(*meshVertTexture) = (UV*)malloc(0);
	face(*meshFace) = (face*)malloc(0);
	vector(*projectedVector) = (vector*)malloc(0);
	RGB(*render)[height] = (RGB(*)[height])malloc(sizeof(RGB) * width * height);
	RGB colorPicker = { .r = 255, .b = 255, .g = 255 };

	FILE* modelInputOBJ = fopen("resources\\test cube.obj", "rt");
	if (modelInputOBJ == NULL) {
		printf("input .obj file not found\n");
		return;
	}
	FILE* renderResult = fopen("resources\\render result.ppm", "wt");

	//initialize ppm
	fprintf(renderResult, "P3\n%d %d\n255\n", width, height);

	//bg color
	for (int y = 0; y < height; y++) {
		for (int x = 0; x < width; x++) {
			render[x][y].r = 50;
			render[x][y].b = 50;
			render[x][y].g = 50;
		}
	}

	//mesh input
	fgets(&inputBuffer, 256, modelInputOBJ);
	while (feof(modelInputOBJ) == 0) {
		switch (inputBuffer[0]) {
		case '#':
			break;
		case 'v' :
			if (inputBuffer[1] == 'n') {
				//vertex normal
				vertNormalCount++;
				if (meshVertNormal = (vector*)realloc(meshVertNormal, sizeof(vector) * vertNormalCount)) {
					sscanf(inputBuffer, "vn %lf %lf %lf", &meshVertNormal[vertNormalCount - 1].x, &meshVertNormal[vertNormalCount - 1].y, &meshVertNormal[vertNormalCount - 1].z);
				}
				else {
					printf("error while reallocating memory for mesh vertex normal info read\n");
					exitSafety(modelInputOBJ, renderResult, render);
					return;
				}
			}
			else if (inputBuffer[1] == 't') {
				//vertex texture coordinates
				vertTextureCount++;
				if (meshVertTexture = (UV*)realloc(meshVertTexture, sizeof(UV) * vertTextureCount)) {
					meshVertTexture[vertTextureCount - 1].v = 0;
					meshVertTexture[vertTextureCount - 1].w = 0;
					sscanf(inputBuffer, "vt %lf %lf %lf", &meshVertTexture[vertTextureCount - 1].u, &meshVertTexture[vertTextureCount - 1].v, &meshVertTexture[vertTextureCount - 1].w);
				}
				else {
					printf("error while reallocating memory for mesh vertex texture info read\n");
					exitSafety(modelInputOBJ, renderResult, render);
					return;
				}
			}
			else {
				//vertex
				vertexCount++;
				if (meshVertex = (vector*)realloc(meshVertex, sizeof(vector) * vertexCount)) {
					sscanf(inputBuffer, "v %lf %lf %lf", &meshVertex[vertexCount - 1].x, &meshVertex[vertexCount - 1].y, &meshVertex[vertexCount - 1].z);
				}
				else {
					printf("error while reallocating memory for mesh vertex info read\n");
					exitSafety(modelInputOBJ, renderResult, render);
					return;
				}
			}
			break;
		case 'o' :
			break;
		case 's' :
			//shade smooth
			sscanf(inputBuffer, "s %d", &shadeSmooth);
			break;
		case 'f' :
			faceCount++;
			if (meshFace = (face*)realloc(meshFace, sizeof(face) * faceCount)) {
				meshFace[faceCount - 1].point = (unsigned int*)malloc(0);
				meshFace[faceCount - 1].texture = (unsigned int*)malloc(0);
				meshFace[faceCount - 1].normal = (unsigned int*)malloc(0);
			}
			else {
				printf("error while reallocating memory for mesh face info read\n");
				exitSafety(modelInputOBJ, renderResult, render);
				return;
			}
			inpBufferCursor = 1;
			indexType = 0;
			while (inputBuffer[inpBufferCursor] == ' ')
				inpBufferCursor++;
			meshFace[faceCount - 1].vertCount = 0;
			while (inputBuffer[inpBufferCursor] != '\0') {
				if (inputBuffer[inpBufferCursor] == '/') {
					indexType++;
					inpBufferCursor++;
				}
				else if (inputBuffer[inpBufferCursor] <= 57 && inputBuffer[inpBufferCursor] >= 48) {
					memset(indexBuffer, 0, sizeof(indexBuffer));
					indexBufferCursor = 0;
					while (inputBuffer[inpBufferCursor] <= 57 && inputBuffer[inpBufferCursor] >= 48) {
						indexBuffer[indexBufferCursor] = inputBuffer[inpBufferCursor];
						indexBufferCursor++;
						inpBufferCursor++;
					}
					if (sscanf(indexBuffer, "%d", &vertAttribute[indexType]) != 1) {
						vertAttribute[indexType] = 0;
					}
				}
				else {
					meshFace[faceCount - 1].vertCount++;
					//vertex position index allocation
					meshFace[faceCount - 1].point = (unsigned int*)realloc(meshFace[faceCount - 1].point, sizeof(unsigned int) * meshFace[faceCount - 1].vertCount);
					if (vertAttribute[0] >= 0)
						meshFace[faceCount - 1].point[meshFace[faceCount - 1].vertCount - 1] = vertAttribute[0];
					else
						meshFace[faceCount - 1].point[meshFace[faceCount - 1].vertCount - 1] = vertexCount + vertAttribute[0] + 1;

					//vertex texture coordinate index allocation
					meshFace[faceCount - 1].texture = (unsigned int*)realloc(meshFace[faceCount - 1].texture, sizeof(unsigned int) * meshFace[faceCount - 1].vertCount);
					if (vertAttribute[1] >= 0)
						meshFace[faceCount - 1].texture[meshFace[faceCount - 1].vertCount - 1] = vertAttribute[1];
					else
						meshFace[faceCount - 1].texture[meshFace[faceCount - 1].vertCount - 1] = vertTextureCount + vertAttribute[1] + 1;

					//vertex normal index allocation
					meshFace[faceCount - 1].normal = (unsigned int*)realloc(meshFace[faceCount - 1].normal, sizeof(unsigned int) * meshFace[faceCount - 1].vertCount);
					if (vertAttribute[2] >= 0)
						meshFace[faceCount - 1].normal[meshFace[faceCount - 1].vertCount - 1] = vertAttribute[2];
					else
						meshFace[faceCount - 1].normal[meshFace[faceCount - 1].vertCount - 1] = vertNormalCount + vertAttribute[2] + 1;
					indexType = 0;
					while (inputBuffer[inpBufferCursor] == ' ')
						inpBufferCursor++;
					if (inputBuffer[inpBufferCursor] == '\n') {
						break;
					}
				}
			}
			break;
		default :
			break;
		}
		fgets(&inputBuffer, 256, modelInputOBJ);
	}

	//perspectives ect initialization
	perspDist = cameraScale / (2 * tan(FOV / 2));
	perspLV.x = -1 * perspDist;
	projectedVector = (vector*)realloc(projectedVector, sizeof(vector) * vertexCount);
	for (int i = 0; i < vertexCount; i++) {
		vectorBuffer = vectorScale(perspDist / (meshVertex[i].x + perspDist), vectorSubtract(meshVertex[i], perspLV));
		//vectorBuffer.x - perspDist = error amount
		projectedVector[i].x = vectorBuffer.x;
		projectedVector[i].y = vectorBuffer.y;
		projectedVector[i].z = vectorBuffer.z;
		//drawCircle(render, (int)(vectorBuffer.y * 100) + widthHalf, (int)(vectorBuffer.z * 100) + heightHalf, 1.0, colorPicker);
	}

	for (int i = 0; i < faceCount; i++) {
		for (int j = 0; j < meshFace[i].vertCount - 1; j++) {
			drawLine(render, (int)(projectedVector[meshFace[i].point[j] - 1].y * 100) + widthHalf, (int)(projectedVector[meshFace[i].point[j] - 1].z * 100) + heightHalf, (int)(projectedVector[meshFace[i].point[j + 1] - 1].y * 100) + widthHalf, (int)(projectedVector[meshFace[i].point[j + 1] - 1].z * 100) + heightHalf, colorPicker);
		}
		drawLine(render, (int)(projectedVector[meshFace[i].point[meshFace[i].vertCount - 1] - 1].y * 100) + widthHalf, (int)(projectedVector[meshFace[i].point[meshFace[i].vertCount - 1] - 1].z * 100) + heightHalf, (int)(projectedVector[meshFace[i].point[0] - 1].y * 100) + widthHalf, (int)(projectedVector[meshFace[i].point[0] - 1].z * 100) + heightHalf, colorPicker);
	}

	/*
	for (int i = 0; i < vertexCount; i++) {
		printf("vert: %lf %lf %lf\n", meshVertex[i].x, meshVertex[i].y, meshVertex[i].z);
	}
	for (int i = 0; i < vertNormalCount; i++) {
		printf("vert norm: %lf %lf %lf\n", meshVertNormal[i].x, meshVertNormal[i].y, meshVertNormal[i].z);
	}

	drawLine(render, 10, 12, 50, 72, colorPicker);
	*/

	//writing whats in array 'render' to the actual file
	for (int y = 0; y < height; y++) {
		for (int x = 0; x < width; x++) {
			fprintf(renderResult, "%d %d %d\t", render[x][y].r, render[x][y].g, render[x][y].b);
		}
		fprintf(renderResult, "\n");
	}
	exitSafety(modelInputOBJ, renderResult, render);
	return;
}
