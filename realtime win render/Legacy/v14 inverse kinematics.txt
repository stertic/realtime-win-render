const pi = 3.141592;
const rad = 57.29577; //180 / pi

void drawLine(float xA, float yA, float xB, float yB) {
	float dx = xB - xA;
	float dy = yB - yA;
	float m = dy / dx, temp, incm;
	if (fabsf(m) <= 1) {
		incm = m;
		if (xA < xB) {
			temp = yA;
			for (float i = xA; i <= xB; i++) {
				gotoxy(i, temp);
				putchar(64);
				temp += incm;
			}
		}
		else {
			temp = yB;
			for (float i = xB; i <= xA; i++) {
				gotoxy(i, temp);
				putchar(64);
				temp += incm;
			}
		}
	}
	else {
		incm = dx / dy;
		if (yA < yB) {
			temp = xA;
			for (float i = yA; i <= yB; i++) {
				gotoxy(temp, i);
				putchar(64);
				temp += incm;
			}
		}
		else {
			temp = xB;
			for (float i = yB; i <= yA; i++) {
				gotoxy(temp, i);
				putchar(64);
				temp += incm;
			}
		}
	}
	gotoxy(xA, yA);
	putchar(88);
	gotoxy(xB, yB);
	putchar(88);
}

float squ(float x) {
	return x * x;
}

float dist(float x0, float y0, float x1, float y1) {
	return sqrtf(squ(x0 - x1) + squ(y0 - y1));
}

void angleTrig(float x0, float y0, float xM, float yM, float x1, float y1, float* outDeg) {
	*outDeg = (atanf(fabsf(y0 - yM) / fabsf(x0 - xM)) + atanf(fabsf(y1 - yM) / fabsf(x1 - xM))) * rad;
	return;
}

bool circleIntersect(float x0, float y0, float r0, float x1, float y1, float r1, float* outXY, float polX, float polY) {
	float x2, y2, a, d, h;
	float dx = x1 - x0;
	float dy = y1 - y0;
	d = sqrtf((dx * dx) + (dy * dy));
	if (d > (r0 + r1)) //no intersection
		return true;
	else if (fabsf(r1 - r0) > d) //inside
		return true;
	a = ((r0 * r0) - (r1 * r1) + (d * d)) / (2.0 * d);
	h = sqrtf((r0 * r0) - (a * a));
	x2 = x0 + ((a * dx) / d);
	y2 = y0 + ((a * dy) / d);
	float tx = dx * (h / d);
	float ty = dy * (h / d);
	if (dist(polX, polY, x2 + ty, y2 - tx) < dist(polX, polY, x2 - ty, y2 + tx)) {
		*outXY = x2 + ty;
		*(outXY + 1) = y2 - tx;
	}
	else {
		*outXY = x2 - ty;
		*(outXY + 1) = y2 + tx;
	}
	return false;
}
