#define resolutionX 1280
#define resolutionY 720
//range of canvas coordinates in code: 0 ~ width - 1, 0 ~ height - 1

#define reportRuntime

//LV = location vector
//RV = rotation vector, could be unit vector or not a unit vector
//all vectors are stored in double type
//all angles are done in radians
//for pixel coordinates, higher x = right, higher y = down
//ID starts from 0
//Count starts from 1

typedef struct {
	uint8_t r, g, b;
} RGB;

typedef struct {
	int x, y;
} XY;

typedef struct {
	double u, v, w;
} UV;

typedef struct {
	double x, y, z;
} vector3;

typedef struct {
	double x, y, z, w;
} vector4;

typedef struct {
	double m[4][4];
} matrix44;

const vector3 zeroVector3 = { .x = 0, .y = 0, .z = 0 };

//data inside struct 'face' are all indices, NOT LOCATION VECTORS
//all are pointers pointing to the start of an array of indices
//negative indices are interpreted into a positive index at reading stage
//indices start at 1, 0 means unused
typedef struct {
	unsigned int pointID[3], textureID[3], normalID[3];
	//the indices are in order
	//ex) vert normal for point[i] is normal[i]
	vector3 faceNormal, geometryCenter;
} tri;

//the three 'Data' pointers points to an array each(vertexData, vertNormalData, vertTextureData)
//data inside the data arrays contain actual position vectors
typedef struct {
	char name[100];
	uint8_t shadeSmooth;
	unsigned int triCount, vertCount, vertNormalCount, vertTextureCount;
	vector3* vertData, * vertNormalData;
	UV* vertTextureData;
	tri* mesh;
} object;

typedef struct {
	unsigned int objCount;
	object* objects;
} scene;

int main() {
	RGB fillCol = { .r = 100, .g = 100, .b = 100 }, bgCol = { .r = 50, .g = 50, .b = 50 };
	XY resolution = { .x = resolutionX, .y = resolutionY };
	scene projScene;
	initScene(&projScene);
	
#ifdef reportRuntime
	double startCLOCK, objParseCLOCK, endCLOCK;
	startCLOCK = (double)(clock()) / CLOCKS_PER_SEC;
#endif

	FILE* renderImg = fopen("Output\\render.ppm", "wt");
	//casting for grid
	RGB** grid = callocArray2D(resolution);
	if (grid == NULL) {
		goto freeMem;
	}

	parseObj("Resource\\input.obj", &(projScene.objects), &(projScene.objCount));

#ifdef reportRuntime
	objParseCLOCK = (double)(clock()) / CLOCKS_PER_SEC;
	printf("obj parsing runtime: %lf\n", objParseCLOCK - startCLOCK);
#endif

	initPlane(&grid, resolution, bgCol);
	initPPM(renderImg, resolution);
	writePPM(renderImg, grid, resolution);

freeMem: {
	freeScene(&projScene);
	free(grid);
	fclose(renderImg);
#ifdef reportRuntime
	endCLOCK = (double)(clock()) / CLOCKS_PER_SEC;
	printf("total runtime: %lf\n", endCLOCK - startCLOCK);
#endif
	}
	return 0;
}

void initPPM(FILE* file, XY res) {
	fprintf(file, "P3\n%d %d\n255\n", res.x, res.y);
	return;
}

inline static void fprintfPPM(FILE* file, int val) {
	if (val / 100 != 0) {
		fputc((val / 100) + 48, file);
		val %= 100;
	}
	if (val / 10 != 0) {
		fputc((val / 10) + 48, file);
		val %= 10;
	}
	fputc((val % 10) + 48, file);
	fputc(32, file);
	return;
}

void writePPM(FILE* file, RGB** grid, XY res) {
	for (int row = 0; row < res.y; row++) {
		for (int col = 0; col < res.x; col++) {
			fprintfPPM(file, grid[col][row].r);
			fprintfPPM(file, grid[col][row].g);
			fprintfPPM(file, grid[col][row].b);
		}
		fputc(10, file);
	}
	return;
}

RGB** callocArray2D(XY res) {
	RGB* castPtr = (RGB*)calloc(res.x * res.y, sizeof(RGB));
	if (castPtr == NULL) {
		printf("callocArray2D : failed to allocate memory\n");
		return NULL;
	}
	RGB** root = (RGB**)calloc(res.x, sizeof(RGB*));
	if (root == NULL) {
		printf("callocArray2D : failed to allocate memory\n");
		return NULL;
	}
	for (int col = 0; col < res.x; col++) {
		root[col] = &castPtr[col * res.y];
	}
	return root;
}

void initPlane(RGB*** inpGrid, XY res, RGB fillCol) {
	for (int row = 0; row < res.y; row++) {
		for (int col = 0; col < res.x; col++) {
			(*inpGrid)[col][row] = fillCol;
		}
	}
	return;
}

void initScene(scene* inpScene) {
	inpScene->objCount = 0;
	inpScene->objects = NULL;
	return;
}

void freeScene(scene* inpScene) {
#pragma warning( push )
#pragma warning( disable : 6001 )
	for (unsigned int tarObj = 0; tarObj < inpScene->objCount; tarObj++) {
		free((inpScene->objects[tarObj]).vertData);
		free((inpScene->objects[tarObj]).vertNormalData);
		free((inpScene->objects[tarObj]).vertTextureData);
		free((inpScene->objects[tarObj]).mesh);
	}
#pragma warning( pop )
	return;
}

//basic vector arithmatic function
//adds two given vectors
vector3 vector3Add(vector3 vectA, vector3 vectB) {
	vector3 calcResult;
	calcResult.x = vectA.x + vectB.x;
	calcResult.y = vectA.y + vectB.y;
	calcResult.z = vectA.z + vectB.z;
	return calcResult;
}

//basic vector arithmatic function
//subtracts by the second vector from the first vector
vector3 vector3Subtract(vector3 vectA, vector3 vectB) {
	vector3 calcResult;
	calcResult.x = vectA.x - vectB.x;
	calcResult.y = vectA.y - vectB.y;
	calcResult.z = vectA.z - vectB.z;
	return calcResult;
}

//basic vector arithmatic function
//scales a given vector by a scalar
vector3 vector3Scale(double scalar, vector3 vect) {
	vector3 calcResult;
	calcResult.x = scalar * vect.x;
	calcResult.y = scalar * vect.y;
	calcResult.z = scalar * vect.z;
	return calcResult;
}

double vector3Dot(vector3 vectA, vector3 vectB) {
	return (vectA.x * vectB.x) + (vectA.y * vectB.y) + (vectA.z * vectB.z);
}

//draws line on a 2 dimensional RGB grid. takes screen space coordinates as input
//last pixel isn't drawn to lessen overlaps
void drawLine(RGB** grid, XY bounds, XY pointA, XY pointB, RGB lineCol) {
	int rove, cursor;
	//cursor glides through the chosen axis, couple is the other axis.
	float couple, slope;
	XY min, max;
	//slope is RELATIVE TO MAIN AXIS! it's not always dy/dx
	if (pointA.x == pointB.x) {
		if (pointA.y < pointB.y) {
			for (rove = pointA.y; rove < pointB.y; rove++)
				grid[pointA.x][rove] = lineCol;
		}
		else {
			for (rove = pointB.y; rove < pointA.y; rove++)
				grid[pointA.x][rove] = lineCol;
		}
	}
	else if (pointA.y == pointB.y) {
		if (pointA.x < pointB.x) {
			for (rove = pointA.x; rove < pointB.x; rove++)
				grid[rove][pointA.y] = lineCol;
		}
		else {
			for (rove = pointB.x; rove < pointA.x; rove++)
				grid[rove][pointA.y] = lineCol;
		}
	}
	else {
		if (fabsf((float)pointA.y - (float)pointB.y) / fabsf((float)pointA.x - (float)pointB.x) > 1) {
			//cursor advances in Y direction
			if (pointA.y < pointB.y) {
				min = pointA;
				max = pointB;
			}
			else {
				min = pointB;
				max = pointA;
			}
			cursor = min.y;
			couple = (float)min.x;
			slope = ((float)min.x - (float)max.x) / ((float)min.y - (float)max.y);
			while (cursor < max.y) {
				if (couple >= 0 && couple < bounds.x && cursor >= 0 && cursor < bounds.y) {
					//checks if couple and cursor is within 0 ~ width, 0 ~ height before drawing
					grid[(int)couple][cursor] = lineCol;
				}
				cursor++;
				couple += slope;
			}
		}
		else {
			//cursor advances in X direction
			if (pointA.x < pointB.x) {
				min = pointA;
				max = pointB;
			}
			else {
				min = pointB;
				max = pointA;
			}
			cursor = min.x;
			couple = (float)min.y;
			slope = ((float)min.y - (float)max.y) / ((float)min.x - (float)max.x);
			while (cursor < max.x) {
				if (cursor >= 0 && cursor < bounds.x && couple >= 0 && couple < bounds.y) {
					grid[cursor][(int)couple] = lineCol;
				}
				cursor++;
				couple += slope;
			}
		}
	}
	return;
}

//takes the root of an object array and reallocates it to concatanate new objects
//count of newly parsed object is added to objCount
//the int return is only for error states
int parseObj(const char* filename, object** objArr, int* objCount) {
	FILE* inputObj = fopen(filename, "rt");
	if (inputObj == NULL) {
		printf("failed to open .obj file(first read)\n");
		fclose(inputObj);
		return -1;
	}

	void* heapMemPtr;
	int argCount, indexType, indexBufferCursor, inpBufferCursor, vertAttribute[3], castIntToBool;
	const int initialObjCount = (*objCount);
	char inputBuffer[256], indexBuffer[32];
	vector3 castVector3;
	UV castUV;
	object* currentObj = &(*objArr)[*objCount];
	tri* currentTri = NULL;

	fgets(inputBuffer, 256, inputObj);
	while (feof(inputObj) == 0) {
		switch (inputBuffer[0]) {
		case 'v':
			if (inputBuffer[1] == 'n') {
				currentObj->vertNormalCount++;
			}
			else if (inputBuffer[1] == 't') {
				currentObj->vertTextureCount++;
			}
			else {
				currentObj->vertCount++;
			}
			break;
		case 'f':
			currentObj->triCount++;
			break;
		case 'o':
			//object
			(*objCount)++;
			heapMemPtr = realloc(*objArr, sizeof(object) * (*objCount));
			if (heapMemPtr == NULL) {
				printf("error while allocating memory for sceneObjects\n");
				fclose(inputObj);
				return -1;
			}
			*objArr = (object*)heapMemPtr;
			currentObj = &((*objArr)[(*objCount) - 1]);
			currentObj->triCount = 0;
			currentObj->shadeSmooth = 0;
			currentObj->mesh = NULL;
			strcpy(currentObj->name, inputBuffer);
			currentObj->vertCount = 0;
			currentObj->vertNormalCount = 0;
			currentObj->vertTextureCount = 0;
			currentObj->vertData = NULL;
			currentObj->vertNormalData = NULL;
			currentObj->vertTextureData = NULL;
			break;
		case 's':
			//shade smooth
			argCount = sscanf(inputBuffer, "s %d", &castIntToBool);
			currentObj->shadeSmooth = castIntToBool;
			if (argCount != 1) {
				printf("error while reading shade smooth options from provided obj file. the file may be damaged or corrupted.\n");
				fclose(inputObj);
				return -1;
			}
			break;
		default:
			break;
		}
		fgets(inputBuffer, 256, inputObj);
	}
	fclose(inputObj);

	for (int tarObj = initialObjCount; tarObj < (*objCount); tarObj++) {
		currentObj = &((*objArr)[tarObj]);
		heapMemPtr = calloc(currentObj->vertNormalCount, sizeof(vector3));
		if (heapMemPtr == NULL) {
			printf("error while allocating memory for mesh vertex normal info read.\n");
			fclose(inputObj);
			return -1;
		}
		currentObj->vertNormalData = (vector3*)heapMemPtr;

		heapMemPtr = calloc(currentObj->vertTextureCount, sizeof(vector3));
		if (heapMemPtr == NULL) {
			printf("error while allocating memory for mesh vertex texture info read.\n");
			fclose(inputObj);
			return -1;
		}
		currentObj->vertTextureData = (UV*)heapMemPtr;

		heapMemPtr = calloc(currentObj->vertCount, sizeof(vector3));
		if (heapMemPtr == NULL) {
			printf("error while allocating memory for mesh vertex info read\n");
			fclose(inputObj);
			return -1;
		}
		currentObj->vertData = (vector3*)heapMemPtr;

		heapMemPtr = calloc(currentObj->triCount, sizeof(tri));
		if (heapMemPtr == NULL) {
			printf("error while allocating memory for mesh face info read\n");
			fclose(inputObj);
			return -1;
		}
		currentObj->mesh = (tri*)heapMemPtr;
	}

	inputObj = fopen(filename, "rt");
	if (inputObj == NULL) {
		printf("failed to open .obj file(second read)\n");
		fclose(inputObj);
		return -1;
	}

	int triNum = 0, triVertNum = 0, objNum = initialObjCount, vertNormalNum = 0, vertTextureNum = 0, vertNum = 0;
	currentObj = &(*objArr)[initialObjCount];

	fgets(inputBuffer, 256, inputObj);
	while (feof(inputObj) == 0) {
		switch (inputBuffer[0]) {
		case 'v':
			if (inputBuffer[1] == 'n') {
				//vertex normal
				argCount = sscanf(inputBuffer, "vn %lf %lf %lf", &castVector3.x, &castVector3.y, &castVector3.z);
				if (argCount != 3) {
					printf("error while reading vertex normals from provided obj file. the file may be damaged or corrupted.\n");
					fclose(inputObj);
					return -1;
				}
				currentObj->vertNormalData[vertNormalNum] = castVector3;
				vertNormalNum++;
			}
			else if (inputBuffer[1] == 't') {
				//vertex texture coordinates
				argCount = sscanf(inputBuffer, "vt %lf %lf %lf", &castUV.u, &castUV.v, &castUV.w);
				if (argCount == 1) {
					castUV.v = 0;
					castUV.w = 0;
				}
				else if (argCount == 2) {
					castUV.w = 0;
				}
				else if (argCount != 3) {
					printf("error while reading vertex texture coordinates from provided obj file. the file may be damaged or corrupted.\n");
					fclose(inputObj);
					return -1;
				}
				currentObj->vertTextureData[vertTextureNum] = castUV;
				vertTextureNum++;
			}
			else {
				//vertex
				argCount = sscanf(inputBuffer, "v %lf %lf %lf", &castVector3.x, &castVector3.y, &castVector3.z);
				if (argCount != 3) {
					printf("error while reading vertex coordinates from provided obj file. the file may be damaged or corrupted.\n");
					fclose(inputObj);
					return -1;
				}
				currentObj->vertData[vertNum] = castVector3;
				vertNum++;
			}
			break;
		case 'f':
			//face
			//initializations
			currentTri = &(currentObj->mesh[triNum]);
			currentTri->geometryCenter = zeroVector3;
			currentTri->faceNormal = zeroVector3;
			inpBufferCursor = 1;
			indexType = 0;
			while (inputBuffer[inpBufferCursor] == ' ')
				inpBufferCursor++;
			//if number wasn't read, represents it unused by leaving the values 0
			vertAttribute[0] = 0;
			vertAttribute[1] = 0;
			vertAttribute[2] = 0;
			triVertNum = 0;
			//start of proper parsing of current line
			while (inputBuffer[inpBufferCursor] != '\0') {
				//check if cursor is at dividing character
				if (inputBuffer[inpBufferCursor] == '/') {
					indexType++;
					inpBufferCursor++;
				}
				//check if cursor is at a number
				else if (inputBuffer[inpBufferCursor] <= 57 && inputBuffer[inpBufferCursor] >= 48) {
					//manual integer read to buffer, and then read with sscanf for stability
					memset(indexBuffer, 0, sizeof(indexBuffer));
					indexBufferCursor = 0;
					while (inputBuffer[inpBufferCursor] <= 57 && inputBuffer[inpBufferCursor] >= 48) {
						indexBuffer[indexBufferCursor] = inputBuffer[inpBufferCursor];
						indexBufferCursor++;
						inpBufferCursor++;
					}
					argCount = sscanf(indexBuffer, "%d", &vertAttribute[indexType]);
					if(argCount != 1) {
						printf("erroneous data in index buffer while parsing obj file. the file may be damaged or corrupted.\n");
						fclose(inputObj);
						return -1;
					}
					if (indexType == 0) {
						//if index type is vertex position
						currentTri->geometryCenter = vector3Add(currentTri->geometryCenter, currentObj->vertData[vertAttribute[indexType] - 1]);
					}
					else if (indexType == 2) {
						//if index type is vertex normal
						currentTri->faceNormal = vector3Add(currentTri->faceNormal, currentObj->vertNormalData[vertAttribute[indexType] - 1]);
					}
				}
				else {
					currentTri->pointID[triVertNum] = vertAttribute[0];
					currentTri->textureID[triVertNum] = vertAttribute[1];
					currentTri->normalID[triVertNum] = vertAttribute[2];
					indexType = 0;
					triVertNum++;
					inpBufferCursor++;
					while (inputBuffer[inpBufferCursor] == ' ')
						inpBufferCursor++;
				}
			}
			//wavefront faces are in order of location - texture - normals
			currentTri->geometryCenter = vector3Scale((double)(1 / 3), currentTri->geometryCenter);
			currentTri->faceNormal = vector3Scale((double)(1 / 3), currentTri->faceNormal);
			triNum++;
			break;
		case 'o':
			vertNormalNum = 0;
			vertTextureNum = 0;
			vertNum = 0;
			currentObj = &(*objArr)[objNum];
			objNum++;
			triNum = 0;
			break;
		default:
			break;
		}
		fgets(inputBuffer, 256, inputObj);
	}
	fclose(inputObj);
	return 0;
}

//this feature is mostly implemented for debugging purposes.
//it is not intended to be used in final product.
void renderLine(RGB** grid, XY bounds, double perspDist, vector3 vertexA, vector3 vertexB, vector3 camTransformvector3, vector3 perspLV, RGB colorPicker) {
	vector3 projectedvector3A, projectedvector3B;
	XY screenA, screenB;
	int widthHalf = (int)(bounds.x / 2), heightHalf = (int)(bounds.y / 2);
	projectedvector3A = vector3Scale(-1 * perspDist / (vertexA.z - perspDist), vector3Subtract(vector3Add(vertexA, camTransformvector3), perspLV));
	projectedvector3B = vector3Scale(-1 * perspDist / (vertexB.z - perspDist), vector3Subtract(vector3Add(vertexB, camTransformvector3), perspLV));
	screenA.x = abs((int)(projectedvector3A.x * 100 + widthHalf));
	screenA.y = abs((int)(projectedvector3A.y * 100 + heightHalf));
	screenB.x = abs((int)(projectedvector3B.x * 100 + widthHalf));
	screenB.y = abs((int)(projectedvector3B.y * 100 + heightHalf));
	drawLine(grid, bounds, screenA, screenB, colorPicker);
	return;
}

//this feature is mostly implemented for debugging purposes.
//it is not intended to be used in final product.
void renderPoint(RGB** grid, XY bounds, double perspDist, vector3 vertex, vector3 camTransformvector3, vector3 perspLV, RGB colorPicker) {
	vector3 projectedvector3;
	int screenX, screenY;
	int widthHalf = bounds.x / 2, heightHalf = bounds.y / 2;
	projectedvector3 = vector3Scale(-1 * perspDist / (vertex.z - perspDist), vector3Subtract(vector3Add(vertex, camTransformvector3), perspLV));
	screenX = (int)(projectedvector3.x * 100 + widthHalf);
	screenY = (int)(projectedvector3.y * 100 + heightHalf);
	grid[screenX][screenY] = colorPicker;
	return;
}

void inspectScene(scene inpScene) {
	for (unsigned int tarObj = 0; tarObj < inpScene.objCount; tarObj++) {
		printf("object: %s\n", inpScene.objects[tarObj].name);
		if (inpScene.objects[tarObj].shadeSmooth)
			printf("shadesmooth: YES\n");
		else
			printf("shadesmooth: NO\n");
		for (unsigned int tarFace = 0; tarFace < inpScene.objects[tarObj].triCount; tarFace++) {
			printf("F%d\n", tarFace);
			for (unsigned int tarVert = 0; tarVert < 3; tarVert++) {
				printf("\tvert%d: ", tarVert);
				printf("p%d ", inpScene.objects[tarObj].mesh[tarFace].pointID[tarVert]);
				printf("n%d ", inpScene.objects[tarObj].mesh[tarFace].normalID[tarVert]);
				printf("t%d\n", inpScene.objects[tarObj].mesh[tarFace].textureID[tarVert]);
			}
			printf("\n");
		}
		for (unsigned int nowIndex = 0; nowIndex < inpScene.objects->vertCount; nowIndex++)
			printf("P%d: %lf %lf %lf\n", nowIndex, inpScene.objects->vertData[nowIndex].x, inpScene.objects->vertData[nowIndex].y, inpScene.objects->vertData[nowIndex].z);
		for (unsigned int nowIndex = 0; nowIndex < inpScene.objects->vertNormalCount; nowIndex++)
			printf("N%d: %lf %lf %lf\n", nowIndex, inpScene.objects->vertNormalData[nowIndex].x, inpScene.objects->vertNormalData[nowIndex].y, inpScene.objects->vertNormalData[nowIndex].z);
		for (unsigned int nowIndex = 0; nowIndex < inpScene.objects->vertTextureCount; nowIndex++)
			printf("T%d: %lf %lf %lf\n", nowIndex, inpScene.objects->vertTextureData[nowIndex].u, inpScene.objects->vertTextureData[nowIndex].v, inpScene.objects->vertTextureData[nowIndex].w);

	}
	return;
}

void printPlane(RGB** grid, XY res) {
	for (int row = 0; row < res.y; row++) {
		for (int col = 0; col < res.x; col++) {
			printf("(%d %d %d) ", grid[col][row].r, grid[col][row].g, grid[col][row].b);
		}
		printf("\n");
	}
	return;
}
