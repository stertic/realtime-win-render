//data inside struct 'face' are all indices, NOT LOCATION VECTORS
//all are pointers pointing to the start of an array of indices
//negative indices are interpreted into a positive index at reading stage
//indices start at 1, 0 means unused
typedef struct {
	//normals represented with counterclockwise vertex

	//the indices are in order
	//ex) vert normal for point[i] is normal[i]

} tri;

//the three 'Data' pointers points to an array each(vertexData, vertNormalData, vertTextureData)
//data inside the data arrays contain actual position vectors
typedef struct {
	char name[32];
	bool shadeSmooth;
	unsigned int triCount, vertCount, vertNormalCount, vertTextureCount;
	//original vert data, must not be modified when rendering(only modify to modify the mesh permanently)
	vect4* vertData;
	vect3* vertNormalData;
	UV* vertTextureData;
	tri* mesh;
} object;

//assumed symmeteric frustum
typedef struct {
	//frustum properties
	//distN is distance from camera's location to the near projection plane
	//distF is distance from camera's location to the far projection plane
	//width is half of the width of the near projection plane
	//height is half of the height of the near projection plane
	double distN, distF, width, height;
	rad pitch, yaw, roll;
	//camera location vector and lookat target vector
	//lookat target vector is calculated with the camera positioned at the origin
	vect4 LVect, TVect;
	//horizontal field of view
	rad hFOV;
} camera;

typedef struct {
	unsigned int vertCount, triCount;
	vect4* vertData;
	tri* mesh;
} renderObj;

typedef struct {
	camera cam;
	renderObj render;
	unsigned int objCount;
	object* objects;
	vect4* objectPos;
} scene;

int createRenderObj(scene& inpScene) {
	void* castPtr;
	for (int tarObj; tarObj < inpScene.objCount; tarObj++) {
		inpScene.render.vertCount += inpScene.objects[tarObj].vertCount;
		inpScene.render.triCount += inpScene.objects[tarObj].triCount;
	}
	castPtr = malloc(sizeof(vect4) * inpScene.render.vertCount);
	if (castPtr == NULL) {
		printf("createRenderObj: failed to allocate memory for vertData\n");
		return -1;
	}
	inpScene.render.vertData = (vect4*)castPtr;

	castPtr = malloc(sizeof(tri) * inpScene.render.triCount);
	if (castPtr == NULL) {
		printf("createRenderObj: failed to allocate memory for mesh\n");
		return -1;
	}
	inpScene.render.mesh = (tri*)castPtr;
	return 0;
}

//openBLinit must be called beforehand
//called after cam.roll / pitch / yaw is changed externally
int openBLupdate(HDC hdc, scene* inpScene, XY resolution) {
	//cam___ variables are only cast for code conciseness - replace if needed
	double camRoll = inpScene->cam.roll, camPitch = inpScene->cam.pitch, camYaw = inpScene->cam.yaw;

	//rotation matrix(pre-transpose)
	matrix44 viewRotMatrix = { {
		cos(camRoll) * cos(camYaw) - sin(camRoll) * sin(camPitch) * sin(camYaw)	, -1 * sin(camRoll) * cos(camPitch)	, cos(camRoll) * sin(camYaw) + sin(camRoll) * sin(camPitch) * cos(camYaw)	, 0,
		sin(camRoll) * cos(camYaw) + cos(camRoll) * sin(camPitch) * sin(camYaw)	, cos(camRoll) * cos(camPitch)		, sin(camRoll) * sin(camYaw) - cos(camRoll) * sin(camPitch) * cos(camYaw)	, 0,
		-1 * cos(camPitch) * sin(camYaw)										, sin(camPitch)						, cos(camPitch) * cos(camYaw)												, 0,
		0																		, 0									, 0																			, 1 } };

	inpScene->cam.TVect = vect4MultM44({ 0, 1, 0, 1 }, viewRotMatrix);
	printDebug(hdc, L"[ %f %f %f %f ]", inpScene->cam.TVect.x, inpScene->cam.TVect.y, inpScene->cam.TVect.z, inpScene->cam.TVect.w);

	viewRotMatrix = transposeM44(viewRotMatrix);

	matrix44 viewTransMatrix = { {
		1, 0, 0, -1 * inpScene->cam.LVect.x,
		0, 1, 0, -1 * inpScene->cam.LVect.y,
		0, 0, 1, -1 * inpScene->cam.LVect.z,
		0, 0, 0, 1 } };

	matrix44 modelViewMatrix = multM44(viewRotMatrix, viewTransMatrix);

	matrix44 projectionMatrix = { {
		inpScene->cam.distN / inpScene->cam.width, 0, 0, 0,
		0, inpScene->cam.distN / inpScene->cam.height, 0, 0,
		0, 0, -1, -2 * inpScene->cam.distN,
		0, 0, -1, 0 } };

	unsigned int vertDataCount = 0;
	for (unsigned int tarObj = 0; tarObj < inpScene->objCount; tarObj++) {
		object* curObj = &inpScene->objects[tarObj];
		if (inpScene->render.vertData == NULL or inpScene->render.mesh == NULL)
			createRenderObj(*inpScene);
		objectM44Transform(curObj->vertData, &(inpScene->render.vertData[vertDataCount]), modelViewMatrix, curObj->vertCount);
		objectM44Transform(&(inpScene->render.vertData[vertDataCount]), NULL, projectionMatrix, curObj->vertCount);
		vertDataCount += curObj->vertCount;

		//dot product culling performed before clipspace culling
		for (unsigned int tarTri = 0; tarTri < inpScene->objects[tarObj].triCount; tarTri++) {
			if (vect3Dot(inpScene->objects[tarObj].mesh[tarTri].faceNormal, vect4Truncate(inpScene->cam.TVect)) < 0) {

			}
		}
		for (unsigned int tarVert = 0; tarVert < inpScene->objects[tarObj].vertCount; tarVert++) {
			if (inpScene->objects[tarObj].vertData[tarVert].x > 1 || inpScene->objects[tarObj].vertData[tarVert].x < -1 ||
				inpScene->objects[tarObj].vertData[tarVert].y > 1 || inpScene->objects[tarObj].vertData[tarVert].y < -1 ||
				inpScene->objects[tarObj].vertData[tarVert].z > 1 || inpScene->objects[tarObj].vertData[tarVert].z < -1) {

			}
		}
	}

	//now in clip space
	for (unsigned int tarObj = 0; tarObj < inpScene->objCount; tarObj++) {
		for (unsigned int tarVert = 0; tarVert < inpScene->objects[tarObj].vertCount; tarVert++) {
			object* curObj = &inpScene->objects[tarObj];
			curObj->TVDA[TVDA::RENDERTVDA][tarVert].x /= curObj->TVDA[TVDA::RENDERTVDA][tarVert].w;
			curObj->TVDA[TVDA::RENDERTVDA][tarVert].y /= curObj->TVDA[TVDA::RENDERTVDA][tarVert].w;
			curObj->TVDA[TVDA::RENDERTVDA][tarVert].z /= curObj->TVDA[TVDA::RENDERTVDA][tarVert].w;
		}
	}

	POINT renderTri[3];
	HBRUSH fillTri = CreateSolidBrush(RGB(250, 250, 250));
	HBRUSH prevBrush = (HBRUSH)SelectObject(hdc, fillTri);

	HPEN lineTri = CreatePen(PS_SOLID, 1, RGB(10, 10, 10));
	HPEN prevPen = (HPEN)SelectObject(hdc, lineTri);

	for (unsigned int tarObj = 0; tarObj < inpScene->objCount; tarObj++) {
		for (unsigned int tarTri = 0; tarTri < inpScene->objects[tarObj].triCount; tarTri++) {
			object* curObj = &inpScene->objects[tarObj];
			renderTri[0].x = (int)(curObj->TVDA[TVDA::RENDERTVDA][curObj->mesh[tarTri].pointID[0] - 1].x * resolution.x + resolution.x / 2);
			renderTri[0].y = (int)(curObj->TVDA[TVDA::RENDERTVDA][curObj->mesh[tarTri].pointID[0] - 1].y * resolution.y + resolution.y / 2);

			renderTri[1].x = (int)(curObj->TVDA[TVDA::RENDERTVDA][curObj->mesh[tarTri].pointID[1] - 1].x * resolution.x + resolution.x / 2);
			renderTri[1].y = (int)(curObj->TVDA[TVDA::RENDERTVDA][curObj->mesh[tarTri].pointID[1] - 1].y * resolution.y + resolution.y / 2);

			renderTri[2].x = (int)(curObj->TVDA[TVDA::RENDERTVDA][curObj->mesh[tarTri].pointID[2] - 1].x * resolution.x + resolution.x / 2);
			renderTri[2].y = (int)(curObj->TVDA[TVDA::RENDERTVDA][curObj->mesh[tarTri].pointID[2] - 1].y * resolution.y + resolution.y / 2);
			//printDebug(hdc, L"[ %d, %d ] [ %d, %d ] [ %d, %d ]", renderTri[0].x, renderTri[0].y, renderTri[1].x, renderTri[1].y, renderTri[2].x, renderTri[2].y);
			Polygon(hdc, renderTri, 3);
		}
	}

	SelectObject(hdc, prevBrush);
	DeleteObject(fillTri);
	SelectObject(hdc, prevPen);
	DeleteObject(lineTri);
	return 0;
}

	HWND hwnd = CreateWindowEx(
		0, //optional window style

		return -1;

	while (GetMessage(&msg, NULL, 0, 0) > 0) {
		TranslateMessage(&msg);
		DispatchMessage(&msg);
	}

enum timer {
	REFRESH_TIMER = 1
};

LRESULT CALLBACK winProc(HWND hwnd, UINT uMsg, WPARAM wParam, LPARAM lParam) {
	static HDC bufferHDC;
	static HBITMAP bufferHBM;

	RECT clientRect;
	GetClientRect(hwnd, &clientRect);
	int width = clientRect.right - clientRect.left,
		height = clientRect.bottom - clientRect.top;
	POINT curP;
	GetCursorPos(&curP);

	projScene.cam.yaw = (double)-curP.x / GetSystemMetrics(SM_CXSCREEN);
	projScene.cam.pitch = (double)curP.y / GetSystemMetrics(SM_CYSCREEN);

	switch (uMsg) {
	case WM_CREATE: {
		ShowCursor(FALSE);
		initDoubleBuffer(hwnd, &bufferHDC, &bufferHBM);
		SetTimer(hwnd, timer::REFRESH_TIMER, 10, NULL);
		resolution.x = width;
		resolution.y = height;

		initScene(&projScene);
		parseObj("Resource\\input.obj", &(projScene.objects), &(projScene.objCount));
		break;
	}
	case WM_TIMER :
		if (wParam == 1)
			InvalidateRect(hwnd, NULL, FALSE);
		break;
	case WM_SIZE: {
		resolution.x = width;
		resolution.y = height;

		cleanDoubleBuffer(bufferHDC, bufferHBM);
		initDoubleBuffer(hwnd, &bufferHDC, &bufferHBM);
		break;
	}
	case WM_PAINT: {
		PAINTSTRUCT ps;
		HDC hdc = BeginPaint(hwnd, &ps);

		//drawing background
		HBRUSH bgBrush = CreateSolidBrush(RGB(50, 50, 50));
		RECT bgRect = {
			0,	//left
			0,	//top
			width,	//right
			height	//bottom
		};
		FillRect(bufferHDC, &bgRect, bgBrush);
		DeleteObject(bgBrush);

		//printDebug(bufferHDC, L"cursor x: %d\ncursor y: %d\nwidth: %d\n height: %d", curP.x, curP.y, resolution.x, resolution.y);

		openBLupdate(bufferHDC, &projScene, resolution);

		BitBlt(hdc, 0, 0, width, height, bufferHDC, 0, 0, SRCCOPY);
		EndPaint(hwnd, &ps);
		break;
	}
	case WM_DESTROY:
		cleanDoubleBuffer(bufferHDC, bufferHBM);
		KillTimer(hwnd, timer::REFRESH_TIMER);
		freeScene(&projScene);
		PostQuitMessage(0);
		break;
	case WM_KEYDOWN:
		break;
	case WM_KEYUP:

		break;
	case WM_CHAR:
		break;
	default:
		return DefWindowProc(hwnd, uMsg, wParam, lParam);
		break;
	}
	return 0;
}

void printDebug(HDC hdc, const wchar_t* format, ...) {
	va_list args;
	va_start(args, format);
	RECT textBound = {
			10,	//left
			10,	//top
	};
	int prevBkMode = SetBkMode(hdc, TRANSPARENT), bufferSize = 128;
	COLORREF prevTextCol = SetTextColor(hdc, RGB(255, 255, 255));

	wchar_t* printBuffer = new wchar_t[bufferSize];
	vswprintf_s(printBuffer, bufferSize, format, args);
	DrawText(hdc, printBuffer, -1, &textBound, DT_LEFT | DT_TOP | DT_NOCLIP);
	delete[] printBuffer;

	va_end(args);
	SetBkMode(hdc, prevBkMode);
	SetTextColor(hdc, prevTextCol);
	return;
}

RGB** RGBCalloc2d(XY res) {
	RGB* castPtr = (RGB*)calloc(res.x * res.y, sizeof(RGB));
	if (castPtr == NULL) {
		printf("RGBCalloc2d : failed to allocate memory\n");
		return NULL;
	}
	RGB** root = (RGB**)calloc(res.x, sizeof(RGB*));
	if (root == NULL) {
		printf("RGBCalloc2d : failed to allocate memory\n");
		return NULL;
	}
	for (int col = 0; col < res.x; col++) {
		root[col] = &castPtr[col * res.y];
	}
	return root;
}

	}

}

void initDoubleBuffer(HWND hwnd, HDC *bufferHDC, HBITMAP *bufferHBM) {
	RECT clientRect;
	GetClientRect(hwnd, &clientRect);
	int width = clientRect.right - clientRect.left,
		height = clientRect.bottom - clientRect.top;
	HDC hdc = GetDC(hwnd);
	*bufferHDC = CreateCompatibleDC(hdc);
	*bufferHBM = CreateCompatibleBitmap(hdc, width, height);
	SelectObject(*bufferHDC, *bufferHBM);
	ReleaseDC(hwnd, hdc);
	return;
}

inline void fprintfPPM(FILE* file, int val) {
	if (val / 100 != 0) {
		fputc((val / 100) + 48, file);
		val %= 100;
	}
	if (val / 10 != 0) {
		fputc((val / 10) + 48, file);
		val %= 10;
	}
	fputc((val % 10) + 48, file);
	fputc(32, file);
	return;
}

void cleanDoubleBuffer(HDC bufferHDC, HBITMAP bufferHBM) {
	DeleteObject(bufferHBM);
	DeleteDC(bufferHDC);
	return;
}

void initPlane(RGB*** inpGrid, XY res, RGB fillCol) {
	for (int row = 0; row < res.y; row++) {
		for (int col = 0; col < res.x; col++) {
			(*inpGrid)[col][row] = fillCol;
		}
	}
	return;
}

void initScene(scene* inpScene) {
	inpScene->objCount = 0;
	inpScene->objects = NULL;

	//camera initializaitons
	inpScene->cam.hFOV = 60 * DEGTORAD;
	inpScene->cam.distN = 0.5;
	inpScene->cam.width = tan(inpScene->cam.hFOV) * inpScene->cam.distN;
	inpScene->cam.height = ((double)resolution.y / (double)resolution.x) * inpScene->cam.width;

	inpScene->cam.roll = 0 * DEGTORAD;
	inpScene->cam.pitch = 0 * DEGTORAD;
	inpScene->cam.yaw = 0 * DEGTORAD;
	inpScene->cam.LVect = initVect4;
	inpScene->cam.TVect.x = -1 * sin(inpScene->cam.roll) * cos(inpScene->cam.pitch);
	inpScene->cam.TVect.y = cos(inpScene->cam.roll) * cos(inpScene->cam.pitch);
	inpScene->cam.TVect.z = sin(inpScene->cam.pitch);
	inpScene->cam.TVect.w = 1.0;
	return;
}

void freeScene(scene* inpScene) {
#pragma warning( push )
#pragma warning( disable : 6001 )
	for (unsigned int tarObj = 0; tarObj < inpScene->objCount; tarObj++) {
		free((inpScene->objects[tarObj]).vertData);
		free((inpScene->objects[tarObj]).vertNormalData);
		free((inpScene->objects[tarObj]).vertTextureData);
		free((inpScene->objects[tarObj]).mesh);
		for (unsigned int tarTVDA = 0; tarTVDA < inpScene->objects[tarObj].TVDACount; tarTVDA++) {
			free((inpScene->objects[tarObj]).TVDA[tarTVDA]);
		}
	}
#pragma warning( pop )
	return;
}

//transforms a chunk of vertex location data array with linear transform(matrix multiplication) and stores it into transArr vertex array chunk
//applies transformation directly if NULL is given for desination vertex data array
void objectM44Transform(vect4* inpArr, vect4* transArr, const matrix44 mat, unsigned int vertCount) {
	if (transArr == NULL or transArr == inpArr) {
		vect4* tarDat;
		vect4 castVect4;
		for (unsigned int tarVert = 0; tarVert < vertCount; tarVert++) {
			tarDat = &(inpArr[tarVert]);
			castVect4.x = mat.n[0] * tarDat->x + mat.n[1] * tarDat->y + mat.n[2] * tarDat->z + mat.n[3] * tarDat->w;
			castVect4.y = mat.n[4] * tarDat->x + mat.n[5] * tarDat->y + mat.n[6] * tarDat->z + mat.n[7] * tarDat->w;
			castVect4.z = mat.n[8] * tarDat->x + mat.n[9] * tarDat->y + mat.n[10] * tarDat->z + mat.n[11] * tarDat->w;
			castVect4.w = mat.n[12] * tarDat->x + mat.n[13] * tarDat->y + mat.n[14] * tarDat->z + mat.n[15] * tarDat->w;
			*tarDat = castVect4;
		}
	}
	else {
		vect4* tarDat;
		for (unsigned int tarVert = 0; tarVert < vertCount; tarVert++) {
			tarDat = &(inpArr[tarVert]);
			transArr[tarVert].x = mat.n[0] * tarDat->x + mat.n[1] * tarDat->y + mat.n[2] * tarDat->z + mat.n[3] * tarDat->w;
			transArr[tarVert].y = mat.n[4] * tarDat->x + mat.n[5] * tarDat->y + mat.n[6] * tarDat->z + mat.n[7] * tarDat->w;
			transArr[tarVert].z = mat.n[8] * tarDat->x + mat.n[9] * tarDat->y + mat.n[10] * tarDat->z + mat.n[11] * tarDat->w;
			transArr[tarVert].w = mat.n[12] * tarDat->x + mat.n[13] * tarDat->y + mat.n[14] * tarDat->z + mat.n[15] * tarDat->w;
		}
	}
	return;
}

matrix44 multM44(const matrix44 matA, const matrix44 matB) {
	matrix44 castM44;
	for (int row = 0; row < 4; row++) {
		for (int col = 0; col < 4; col++) {
			castM44.n[row * 4 + col] = matA.n[n(0, row)] * matB.n[n(col, 0)] + matA.n[n(1, row)] * matB.n[n(col, 1)] + matA.n[n(2, row)] * matB.n[n(col, 2)] + matA.n[n(3, row)] * matB.n[n(col, 3)];
		}
	}
	return castM44;
}

matrix44 transposeM44(const matrix44 m) {
	matrix44 cast;
	//0 1 2 3 -> 0 4 8 12
	//4 5 6 7 -> 1 5 9 13
	//8 9 10 11 -> 2 6 10 14
	//12 13 14 15 -> 3 7 11 15

	for (int i = 0; i < 16; i++) {
		cast.n[i] = m.n[i % 4 * 4 + i / 4];
	}
	return cast;
}

//basic vector arithmatic function
//adds two given vectors
vect3 vect3Add(vect3 v1, vect3 v2) {
	vect3 res;
	res.x = v1.x + v2.x;
	res.y = v1.y + v2.y;
	res.z = v1.z + v2.z;
	return res;
}

//basic vector arithmatic function
//subtracts by the second vector from the first vector
vect3 vect3Subt(vect3 v1, vect3 v2) {
	vect3 res;
	res.x = v1.x - v2.x;
	res.y = v1.y - v2.y;
	res.z = v1.z - v2.z;
	return res;
}

//basic vector arithmatic function
//scales a given vector by a scalar
vect3 vect3Scale(double scalar, vect3 v) {
	vect3 res;
	res.x = scalar * v.x;
	res.y = scalar * v.y;
	res.z = scalar * v.z;
	return res;
}

double vect3Dot(vect3 v1, vect3 v2) {
	return (v1.x * v2.x) + (v1.y * v2.y) + (v1.z * v2.z);
}

vect3 vect3Cross(vect3 v1, vect3 v2) {
	vect3 res;
	res.x = v1.y * v2.z - v1.z * v2.y;
	res.y = v1.z * v2.x - v1.x * v2.z;
	res.z = v1.x * v2.y - v1.y - v2.x;
	return res;
}

vect3 vect4Truncate(vect4 v) {
	vect3 cast;
	cast.x = v.x;
	cast.y = v.y;
	cast.z = v.z;
	return cast;
}

vect4 vect4MultM44(vect4 v, matrix44 mat) {
	vect4 castVect4;
	castVect4.x = v.x * mat.n[0] + v.y * mat.n[1] + v.z * mat.n[2] + v.w * mat.n[3];
	castVect4.y = v.x * mat.n[4] + v.y * mat.n[5] + v.z * mat.n[6] + v.w * mat.n[7];
	castVect4.z = v.x * mat.n[8] + v.y * mat.n[9] + v.z * mat.n[10] + v.w * mat.n[11];
	castVect4.w = v.x * mat.n[12] + v.y * mat.n[13] + v.z * mat.n[14] + v.w * mat.n[15];
	return castVect4;
}

//takes the root of an object array and reallocates it to concatanate new objects
//count of newly parsed object is added to objCount
//the int return is only for error states
//indices are global in wavefront formatting, but are parsed to be local to the object
int parseObj(const char* filename, object** objArr, unsigned int* objCount) {
	FILE* inputObj;
	fopen_s(&inputObj, filename, "rt");
	if (inputObj == NULL) {
		printf("failed to open .obj file(first read)\n");
		return -1;
	}

	void* heapMemPtr;
	int argCount, indexType, indexBufferCursor, inpBufferCursor, vertAttribute[3], castIntToBool;
	const unsigned int initialObjCount = (*objCount);
	char inputBuffer[256], indexBuffer[32];
	vect3 castVect3;
	vect4 castVect4 = initVect4;
	UV castUV;
	object* currentObj = &(*objArr)[*objCount];
	tri* currentTri = NULL;

	fgets(inputBuffer, 256, inputObj);
	while (feof(inputObj) == 0) {
		switch (inputBuffer[0]) {
		case 'v':
			if (inputBuffer[1] == 'n') {
				currentObj->vertNormalCount++;
			}
			else if (inputBuffer[1] == 't') {
				currentObj->vertTextureCount++;
			}
			else {
				currentObj->vertCount++;
			}
			break;
		case 'f':
			currentObj->triCount++;
			break;
		case 'o':
			//object
			(*objCount)++;
			heapMemPtr = realloc(*objArr, sizeof(object) * (*objCount));
			if (heapMemPtr == NULL) {
				printf("error while allocating memory for sceneObjects\n");
				fclose(inputObj);
				return -1;
			}
			*objArr = (object*)heapMemPtr;
			currentObj = &((*objArr)[(*objCount) - 1]);
			currentObj->triCount = 0;
			currentObj->shadeSmooth = 0;
			currentObj->mesh = NULL;
			strcpy_s(currentObj->name, inputBuffer);
			currentObj->vertCount = 0;
			currentObj->vertNormalCount = 0;
			currentObj->vertTextureCount = 0;
			currentObj->vertData = NULL;
			currentObj->vertNormalData = NULL;
			currentObj->vertTextureData = NULL;
			break;
		case 's':
			//shade smooth
			argCount = sscanf_s(inputBuffer, "s %d", &castIntToBool);
			currentObj->shadeSmooth = castIntToBool;
			if (argCount != 1) {
				printf("error while reading shade smooth options from provided obj file. the file may be damaged or corrupted.\n");
				fclose(inputObj);
				return -1;
			}
			break;
		default:
			break;
		}
		fgets(inputBuffer, 256, inputObj);
	}
	fclose(inputObj);

	for (unsigned int tarObj = initialObjCount; tarObj < (*objCount); tarObj++) {
		currentObj = &((*objArr)[tarObj]);
		heapMemPtr = calloc(currentObj->vertNormalCount, sizeof(vect3));
		if (heapMemPtr == NULL) {
			printf("error while allocating memory for mesh vertex normal info read.\n");
			fclose(inputObj);
			return -1;
		}
		currentObj->vertNormalData = (vect3*)heapMemPtr;

		heapMemPtr = calloc(currentObj->vertTextureCount, sizeof(vect3));
		if (heapMemPtr == NULL) {
			printf("error while allocating memory for mesh vertex texture info read.\n");
			fclose(inputObj);
			return -1;
		}
		currentObj->vertTextureData = (UV*)heapMemPtr;

		heapMemPtr = calloc(currentObj->vertCount, sizeof(vect4));
		if (heapMemPtr == NULL) {
			printf("error while allocating memory for mesh vertex info read\n");
			fclose(inputObj);
			return -1;
		}
		currentObj->vertData = (vect4*)heapMemPtr;

		heapMemPtr = calloc(currentObj->triCount, sizeof(tri));
		if (heapMemPtr == NULL) {
			printf("error while allocating memory for mesh face info read\n");
			fclose(inputObj);
			return -1;
		}
		currentObj->mesh = (tri*)heapMemPtr;
	}

	fopen_s(&inputObj, filename, "rt");
	if (inputObj == NULL) {
		printf("failed to open .obj file(second read)\n");
		return -1;
	}

	int triNum = 0, triVertNum = 0, vertNormalNum = 0, vertTextureNum = 0, vertNum = 0;
	unsigned int pointIDStart = 0, normalIDStart = 0, textureIDStart = 0, objNum = initialObjCount;
	currentObj = &(*objArr)[initialObjCount];

	fgets(inputBuffer, 256, inputObj);
	while (feof(inputObj) == 0) {
		switch (inputBuffer[0]) {
		case 'v':
			if (inputBuffer[1] == 'n') {
				//vertex normal
				argCount = sscanf_s(inputBuffer, "vn %lf %lf %lf", &castVect3.x, &castVect3.y, &castVect3.z);
				if (argCount != 3) {
					printf("error while reading vertex normals from provided obj file. the file may be damaged or corrupted.\n");
					fclose(inputObj);
					return -1;
				}
				currentObj->vertNormalData[vertNormalNum] = castVect3;
				vertNormalNum++;
			}
			else if (inputBuffer[1] == 't') {
				//vertex texture coordinates
				argCount = sscanf_s(inputBuffer, "vt %lf %lf %lf", &castUV.u, &castUV.v, &castUV.w);
				if (argCount == 1) {
					castUV.v = 0;
					castUV.w = 0;
				}
				else if (argCount == 2) {
					castUV.w = 0;
				}
				else if (argCount != 3) {
					printf("error while reading vertex texture coordinates from provided obj file. the file may be damaged or corrupted.\n");
					fclose(inputObj);
					return -1;
				}
				currentObj->vertTextureData[vertTextureNum] = castUV;
				vertTextureNum++;
			}
			else {
				//vertex
				argCount = sscanf_s(inputBuffer, "v %lf %lf %lf", &castVect4.x, &castVect4.y, &castVect4.z);
				if (argCount != 3) {
					printf("error while reading vertex coordinates from provided obj file. the file may be damaged or corrupted.\n");
					fclose(inputObj);
					return -1;
				}
				currentObj->vertData[vertNum] = castVect4;
				vertNum++;
			}
			break;
		case 'f':
			//face
			//initializations
			currentTri = &(currentObj->mesh[triNum]);
			currentTri->geometryCenter = zeroVect3;
			currentTri->faceNormal = zeroVect3;
			inpBufferCursor = 1;
			indexType = 0;
			while (inputBuffer[inpBufferCursor] == ' ')
				inpBufferCursor++;
			//if number wasn't read, represents it unused by leaving the values 0
			vertAttribute[0] = 0;
			vertAttribute[1] = 0;
			vertAttribute[2] = 0;
			triVertNum = 0;
			//start of proper parsing of current line
			while (inputBuffer[inpBufferCursor] != '\0') {
				//check if cursor is at dividing character
				if (inputBuffer[inpBufferCursor] == '/') {
					indexType++;
					inpBufferCursor++;
				}
				//check if cursor is at a number
				else if (inputBuffer[inpBufferCursor] <= 57 && inputBuffer[inpBufferCursor] >= 48) {
					//manual integer read to buffer, and then read with sscanf for stability
					memset(indexBuffer, 0, sizeof(indexBuffer));
					indexBufferCursor = 0;
					while (inputBuffer[inpBufferCursor] <= 57 && inputBuffer[inpBufferCursor] >= 48) {
						indexBuffer[indexBufferCursor] = inputBuffer[inpBufferCursor];
						indexBufferCursor++;
						inpBufferCursor++;
					}
					argCount = sscanf_s(indexBuffer, "%d", &vertAttribute[indexType]);
					if (argCount != 1) {
						printf("erroneous data in index buffer while parsing obj file. the file may be damaged or corrupted.\n");
						fclose(inputObj);
						return -1;
					}
					if (indexType == 0) {
						//if index type is vertex position
						currentTri->geometryCenter = vect3Add(currentTri->geometryCenter, vect4Truncate(currentObj->vertData[vertAttribute[indexType] - 1]));
					}
					else if (indexType == 2) {
						//if index type is vertex normal
						currentTri->faceNormal = vect3Add(currentTri->faceNormal, currentObj->vertNormalData[vertAttribute[indexType] - 1]);
					}
				}
				else {
					currentTri->pointID[triVertNum] = vertAttribute[0] - pointIDStart;
					currentTri->textureID[triVertNum] = vertAttribute[1] - textureIDStart;
					currentTri->normalID[triVertNum] = vertAttribute[2] - normalIDStart;
					indexType = 0;
					triVertNum++;
					inpBufferCursor++;
					while (inputBuffer[inpBufferCursor] == ' ')
						inpBufferCursor++;
				}
			}
			//wavefront faces are in order of location - texture - normals
			currentTri->geometryCenter = vect3Scale((double)(1 / 3), currentTri->geometryCenter);
			currentTri->faceNormal = vect3Scale((double)(1 / 3), currentTri->faceNormal);
			triNum++;
			break;
		case 'o':
			if (objNum != initialObjCount) {
				pointIDStart += (*objArr)[objNum - 1].vertCount;
				normalIDStart += (*objArr)[objNum - 1].vertNormalCount;
				textureIDStart += (*objArr)[objNum - 1].vertTextureCount;
			}
			vertNormalNum = 0;
			vertTextureNum = 0;
			vertNum = 0;
			currentObj = &(*objArr)[objNum];
			objNum++;
			triNum = 0;
			break;
		default:
			break;
		}
		fgets(inputBuffer, 256, inputObj);
	}
	fclose(inputObj);
	return 0;
}
