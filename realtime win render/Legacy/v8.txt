#define width 1280
#define height 720
//range of canvas coordinates in code: 0 ~ width - 1, 0 ~ height - 1

//LV = location vector
//RV = rotation vector, could be unit vector or not a unit vector
//all vectors are stored in double type
//all angles are done in radians
//for pixel coordinates, higher x = right, higher y = down

//rgb are stored in char because it's 1 byte so it's efficient for storing numbers within 0 ~ 255
//needs to be unsigned or it gives negative rgb values
typedef struct {
	unsigned char r;
	unsigned char b;
	unsigned char g;
} RGB;

//screenspace coordinates
typedef struct {
	unsigned int x;
	unsigned int y;
} XY;

typedef struct {
	double u;
	double v;
	double w;
} UV;

typedef struct {
	double x;
	double y;
	double z;
} vector;

const vector zeroVector = { .x = 0, .y = 0, .z = 0 };

//data inside struct 'face' are all indices, NOT LOCATION VECTORS
//all are pointers pointing to the start of an array of indices
//negative indices are interpreted into a positive index at reading stage
//indices start at 1, 0 means unused.
typedef struct {
	unsigned int vertCount;
	unsigned int* point;
	unsigned int* texture;
	unsigned int* normal;
	vector faceNormal;
	vector geometryCenter;
	//the indices are in order.
	//ex) vert normal for point[i] is normal[i].
} face;

typedef struct {
	char* name;
	unsigned int faceCount;
	bool shadeSmooth;
	face* mesh;
} object;

//the three pointers points to an array each(vertexData, vertNormalData, vertTextureData)
//data inside the arrays contain actual position vectors
typedef struct {
	unsigned int vertCount;
	unsigned int vertNormalCount;
	unsigned int vertTextureCount;
	vector* vertData;
	vector* vertNormalData;
	UV* vertTextureData;
	unsigned int objectCount;
	object* objects;
} scene;

//basic vector arithmatic function
//adds two given vectors
vector vectorAdd(vector vectA, vector vectB) {
	vector calcResult;
	calcResult.x = vectA.x + vectB.x;
	calcResult.y = vectA.y + vectB.y;
	calcResult.z = vectA.z + vectB.z;
	return calcResult;
}

//basic vector arithmatic function
//subtracts by the second vector from the first vector
vector vectorSubtract(vector vectA, vector vectB) {
	vector calcResult;
	calcResult.x = vectA.x - vectB.x;
	calcResult.y = vectA.y - vectB.y;
	calcResult.z = vectA.z - vectB.z;
	return calcResult;
}

//basic vector arithmatic function
//scales a given vector by a scalar
vector vectorScale(double scalar, vector vect) {
	vector calcResult;
	calcResult.x = scalar * vect.x;
	calcResult.y = scalar * vect.y;
	calcResult.z = scalar * vect.z;
	return calcResult;
}

double vectorDot(vector vectA, vector vectB) {
	return (vectA.x * vectB.x) + (vectA.y * vectB.y) + (vectA.z * vectB.z);
}

//slider at 0 returns valA, slider at 1 returns valB
//unclamped
//child function for drawTrigBland
double linInterp(double slider, double valA, double valB) {
	return valA + (valB - valA) * slider;
}

//draws line on a 2 dimensional RGB grid. takes screen space coordinates as input
//last pixel isn't drawn to lessen overlaps
void drawLine(RGB(*drawTo)[height], XY pointA, XY pointB, RGB lineCol) {
	unsigned int rove, cursor;
	//cursor glides through the chosen axis, couple is the other axis.
	float couple, slope;
	XY min, max;
	//slope is RELATIVE TO MAIN AXIS! it's not always dy/dx
	if (pointA.x == pointB.x) {
		if (pointA.y < pointB.y) {
			for (rove = pointA.y; rove < pointB.y; rove++)
				drawTo[pointA.x][rove] = lineCol;
		}
		else {
			for (rove = pointB.y; rove < pointA.y; rove++)
				drawTo[pointA.x][rove] = lineCol;
		}
		
	}
	else if (pointA.y == pointB.y) {
		if (pointA.x < pointB.x) {
			for (rove = pointA.x; rove < pointB.x; rove++)
				drawTo[rove][pointA.y] = lineCol;
		}
		else {
			for (rove = pointB.x; rove < pointA.x; rove++)
				drawTo[rove][pointA.y] = lineCol;
		}
	}
	else {
		if (fabsf((float)pointA.y - (float)pointB.y) / fabsf((float)pointA.x - (float)pointB.x) > 1) {
			//cursor advances in Y direction
			if (pointA.y < pointB.y) {
				min = pointA;
				max = pointB;
			}
			else {
				min = pointB;
				max = pointA;
			}
			cursor = min.y;
			couple = (float)min.x;
			slope = ((float)min.x - (float)max.x) / ((float)min.y - (float)max.y);
			while (cursor < max.y) {
				if (couple >= 0 && couple < width && cursor >= 0 && cursor < height) {
					//checks if couple and cursor is within 0 ~ width, 0 ~ height before drawing
					drawTo[(unsigned int)couple][cursor] = lineCol;
				}
				cursor++;
				couple += slope;
			}
		}
		else {
			//cursor advances in X direction
			if (pointA.x < pointB.x) {
				min = pointA;
				max = pointB;
			}
			else {
				min = pointB;
				max = pointA;
			}
			cursor = min.x;
			couple = (float)min.y;
			slope = ((float)min.y -(float)max.y) / ((float)min.x - (float)max.x);
			while (cursor < max.x) {
				if (cursor >= 0 && cursor < width && couple >= 0 && couple < height) {
					drawTo[cursor][(unsigned int)couple] = lineCol;
				}
				cursor++;
				couple += slope;
			}
		}
	}
	return;
}

//child function of drawTrigBland
//accepts pointers of three for unsorted XY types and sorts them by their x
//pointA -> highest, pointB -> mid, pointC -> lowest
void sortTripleX(XY* pointA, XY* pointB, XY* pointC) {
	XY temp;
	if (pointA->x < pointB->x) {
		temp = *pointA;
		*pointA = *pointB;
		*pointB = temp;
	}
	if (pointA->x < pointC->x) {
		temp = *pointA;
		*pointA = *pointC;
		*pointC = temp;
	}
	if (pointB->x < pointC->x) {
		temp = *pointB;
		*pointB = *pointC;
		*pointC = temp;
	}
	return;
}

//child function of drawTrigBland
//accepts pointers of three for unsorted XY types and sorts them by their y
//pointA -> highest, pointB -> mid, pointC -> lowest
void sortTripleY(XY* pointA, XY* pointB, XY* pointC) {
	XY temp;
	if (pointA->y < pointB->y) {
		temp = *pointA;
		*pointA = *pointB;
		*pointB = temp;
	}
	if (pointA->y < pointC->y) {
		temp = *pointA;
		*pointA = *pointC;
		*pointC = temp;
	}
	if (pointB->y < pointC->y) {
		temp = *pointB;
		*pointB = *pointC;
		*pointC = temp;
	}
	return;
}

//child function of drawTrigBland
//draws a triangle for the lower half(from low to mid) of the triangle
void firstInterpTrigBland(RGB(*drawTo)[height], XY high, XY mid, XY low, RGB fillCol) {
	unsigned int scanRow, column;
	float startScan, endScan, startSlope, endSlope;
	if (mid.x < linInterp((mid.y - low.y) / (high.y - low.y), low.x, high.x)) {
		startScan = (float)low.x;
		endScan = (float)low.x;
		startSlope = (float)(mid.x - low.x) / (float)(mid.y - low.y);
		endSlope = (float)(high.x - low.x) / (float)(high.y - low.y);
		scanRow = low.y;
		while (scanRow < mid.y) {
			for (column = (unsigned int)startScan; column <= (unsigned int)endScan; column++)
				drawTo[column][scanRow] = fillCol;
			//drawLine(drawTo, startScan, scanRow, endScan, scanRow, fillCol);
			//i changed drawline function to use XY. this needs to be changed if it is to be tested.
			//think casting it to some other temp XY and then sending it would work, but there might be better solutions i think
			scanRow++;
			startScan += startSlope;
			endScan += endSlope;
		}
	}
	else {
		startScan = (float)low.x;
		endScan = (float)low.x;
		startSlope = (float)(high.x - low.x) / (float)(high.y - low.y);
		endSlope = (float)(mid.x - low.x) / (float)(mid.y - low.y);
		scanRow = low.y;
		while (scanRow < mid.y) {
			for (column = (unsigned int)startScan; column <= (unsigned int)endScan; column++)
				drawTo[column][scanRow] = fillCol;
			//drawLine(drawTo, startScan, scanRow, endScan, scanRow, fillCol);
			//same goes here
			scanRow++;
			startScan += startSlope;
			endScan += endSlope;
		}
	}
	return;
}

//child function of drawTrigBland
//draws a triangle for the upper half(from mid til high) of the triangle
void secondInterpTrigBland(RGB(*drawTo)[height], XY high, XY mid, XY low, RGB fillCol) {
	unsigned int scanRow, column;
	float startScan, endScan, startSlope, endSlope;
	float interpolate = (float)linInterp(((double)mid.y - (double)low.y) / ((double)high.y - (double)low.y), (double)low.x, (double)high.x);
	scanRow = mid.y;
	if (mid.x < interpolate) {
		startSlope = ((float)high.x - (float)mid.x) / ((float)high.y - (float)mid.y);
		endSlope = ((float)high.x - interpolate) / ((float)high.y - (float)mid.y);
		startScan = (float)mid.x;
		endScan = interpolate;
		while (scanRow < high.y) {
			for (column = (unsigned int)startScan; column < (unsigned int)endScan; column++)
				drawTo[column][scanRow] = fillCol;
			scanRow++;
			startScan += startSlope;
			endScan += endSlope;
		}
	}
	else {
		startSlope = ((float)high.x - interpolate) / ((float)high.y - (float)mid.y);
		endSlope = ((float)high.x - (float)mid.x) / ((float)high.y - (float)mid.y);
		startScan = interpolate;
		endScan = (float)mid.x;
		while (scanRow < high.y) {
			for (column = (unsigned int)startScan; column < (unsigned int)endScan; column++)
				drawTo[column][scanRow] = fillCol;
			scanRow++;
			startScan += startSlope;
			endScan += endSlope;
		}
	}
	return;
}

//side outline pixels are drawn, the last side of triangle isn't drawn to lessen overlaps
void drawTrigBland(RGB(*drawTo)[height], XY pointA, XY pointB, XY pointC, RGB fillCol) {
	XY high = pointA, mid = pointB, low = pointC;
	unsigned int scanRow = low.y, column;
	//sorting in height - linescans scans through the Y axis, always.
	sortTripleY(&high, &mid, &low);
	//edgecases management - preventing division by 0
	//all three points lie on a single line
	if (low.y == high.y) {
		sortTripleX(&high, &mid, &low);
		for (column = low.x; column <= high.x; column++)
			drawTo[column][scanRow] = fillCol;
	}
	//skips first interpolation, draws a line for the first interpolation instead
	else if (mid.y == low.y) {
		drawLine(drawTo, mid, low, fillCol);
		secondInterpTrigBland(drawTo, high, mid, low, fillCol);
	}
	//skips second interpolation, draws a line for the second interpolation instead
	else if (mid.y == high.y) {
		firstInterpTrigBland(drawTo, high, mid, low, fillCol);
		drawLine(drawTo, mid, high, fillCol);
	}
	else {
		firstInterpTrigBland(drawTo, high, mid, low, fillCol);
		secondInterpTrigBland(drawTo, high, mid, low, fillCol);
	}
	return;
}

void drawCircle(RGB(*drawTo)[height], int screenX, int screenY, RGB dotCol) {
	drawTo[screenX][screenY] = dotCol;
	drawTo[screenX + 1][screenY] = dotCol;
	drawTo[screenX - 1][screenY] = dotCol;
	drawTo[screenX][screenY + 1] = dotCol;
	drawTo[screenX][screenY - 1] = dotCol;
	return;
}

void drawFaceOutline(RGB(*drawTo)[height], face targetFace, XY* screenspaceXY, RGB colorPicker) {
	XY screenspaceStart, screenspaceEnd;
	unsigned int screenLastPoint = targetFace.point[targetFace.vertCount - 1];
	for (unsigned int tarVert = 0; tarVert < targetFace.vertCount - 1; tarVert++) {
		screenspaceStart = screenspaceXY[targetFace.point[tarVert] - 1];
		screenspaceEnd = screenspaceXY[targetFace.point[tarVert + 1] - 1];
		drawLine(drawTo, screenspaceStart, screenspaceEnd, colorPicker);
	}
	drawLine(drawTo, screenspaceXY[screenLastPoint - 1], screenspaceXY[targetFace.point[0] - 1], colorPicker);
	return;
}

//function to handle safer dynamic memory reallocation for index arrays
//reallocates data at memBlock then appends data(index) onto it.
void addVertAttribute(unsigned int** memBlock, unsigned int data, unsigned int wrapAround, unsigned int vertCount) {
	unsigned int* tempPtr = (unsigned int*)realloc(*memBlock, sizeof(unsigned int) * vertCount);
	if (tempPtr == NULL) {
		printf("failed to reallocate memory within addVertAttribute.\n! memory was not freed instatly afterwards !\n");
		return;
	}
	*memBlock = tempPtr;
	//wraparound is needed since negative indecies are used in wavefront standards
	if (data >= 0)
		(*memBlock)[vertCount - 1] = data;
	else
		(*memBlock)[vertCount - 1] = wrapAround + data + 1;
	return;
}

//this feature is mostly implemented for debugging purposes.
//it is not intended to be used in final product.
void renderLine(RGB(*drawTo)[height], double perspDist, vector vertexA, vector vertexB, vector camTransformVector, vector perspLV, RGB colorPicker) {
	vector projectedVectorA, projectedVectorB;
	XY screenA, screenB;
	int widthHalf = (int)(width / 2), heightHalf = (int)(height / 2);
	projectedVectorA = vectorScale(-1 * perspDist / (vertexA.z - perspDist), vectorSubtract(vectorAdd(vertexA, camTransformVector), perspLV));
	projectedVectorB = vectorScale(-1 * perspDist / (vertexB.z - perspDist), vectorSubtract(vectorAdd(vertexB, camTransformVector), perspLV));
	screenA.x = abs((int)(projectedVectorA.x * 100 + widthHalf));
	screenA.y = abs((int)(projectedVectorA.y * 100 + heightHalf));
	screenB.x = abs((int)(projectedVectorB.x * 100 + widthHalf));
	screenB.y = abs((int)(projectedVectorB.y * 100 + heightHalf));
	drawLine(drawTo, screenA, screenB, colorPicker);
	return;
}

	vector projectedVector;

	int widthHalf = (int)(width / 2), heightHalf = (int)(height / 2);

int parseObj(FILE* modelInputOBJ, scene* inpScene) {
	if (modelInputOBJ == NULL) {
		return 1;
	}
	void* heapMemPtr;
	int argCount, indexType, indexBufferCursor, inpBufferCursor, vertAttribute[3], castIntToBool;
	char inputBuffer[256], indexBuffer[32];
	vector castVector;
	UV castUV;
	face* currentFace;
	object* currentObj;

	fgets(inputBuffer, 256, modelInputOBJ);
	while (feof(modelInputOBJ) == 0) {
		switch (inputBuffer[0]) {
		case 'v':
			if (inputBuffer[1] == 'n') {
				//vertex normal
				inpScene->vertNormalCount++;
				heapMemPtr = realloc(inpScene->vertNormalData, sizeof(vector) * inpScene->vertNormalCount);
				if (heapMemPtr == NULL) {
					printf("error while reallocating memory for mesh vertex normal info read.\n");
					return 1;
				}
				inpScene->vertNormalData = (vector*)heapMemPtr;
				argCount = sscanf(inputBuffer, "vn %lf %lf %lf", &castVector.x, &castVector.y, &castVector.z);
				if (argCount != 3) {
					printf("error while reading vertex normals from provided obj file. the file may be damaged or corrupted.\n");
					return 1;
				}
				inpScene->vertNormalData[inpScene->vertNormalCount - 1] = castVector;
			}
			else if (inputBuffer[1] == 't') {
				//vertex texture coordinates
				inpScene->vertTextureCount++;
				heapMemPtr = realloc(inpScene->vertTextureData, sizeof(UV) * inpScene->vertTextureCount);
				if (heapMemPtr == NULL) {
					printf("error while reallocating memory for mesh vertex texture info read.\n");
					return 1;
				}
				inpScene->vertTextureData = (UV*)heapMemPtr;
				argCount = sscanf(inputBuffer, "vt %lf %lf %lf", &castUV.u, &castUV.v, &castUV.w);
				if (argCount == 1) {
					castUV.v = 0;
					castUV.w = 0;
				}
				else if (argCount == 2) {
					castUV.w = 0;
				}
				else if (argCount != 3) {
					printf("error while reading vertex texture coordinates from provided obj file. the file may be damaged or corrupted.\n");
					return 1;
				}
				inpScene->vertTextureData[inpScene->vertTextureCount - 1] = castUV;
			}
			else {
				//vertex
				inpScene->vertCount++;
				heapMemPtr = realloc(inpScene->vertData, sizeof(vector) * inpScene->vertCount);
				if (heapMemPtr == NULL) {
					printf("error while reallocating memory for mesh vertex info read\n");
					return 1;
				}
				inpScene->vertData = (vector*)heapMemPtr;
				argCount = sscanf(inputBuffer, "v %lf %lf %lf", &castVector.x, &castVector.y, &castVector.z);
				if (argCount != 3) {
					printf("error while reading vertex coordinates from provided obj file. the file may be damaged or corrupted.\n");
					return 1;
				}
				inpScene->vertData[inpScene->vertCount - 1] = castVector;
			}
			break;
		case 'f':
			//face
			inpScene->objects[inpScene->objectCount - 1].faceCount++;
			heapMemPtr = realloc(inpScene->objects[inpScene->objectCount - 1].mesh, sizeof(face) * inpScene->objects[inpScene->objectCount - 1].faceCount);
			if (heapMemPtr == NULL) {
				printf("error while reallocating memory for mesh face info read\n");
				return 1;
			}
			inpScene->objects[inpScene->objectCount - 1].mesh = (face*)heapMemPtr;

			//initializations
			currentFace = &(inpScene->objects[inpScene->objectCount - 1].mesh[inpScene->objects[inpScene->objectCount - 1].faceCount - 1]);
			currentFace->point = NULL;
			currentFace->texture = NULL;
			currentFace->normal = NULL;
			currentFace->geometryCenter = zeroVector;
			currentFace->faceNormal = zeroVector;
			inpBufferCursor = 1;
			indexType = 0;
			while (inputBuffer[inpBufferCursor] == ' ')
				inpBufferCursor++;
			currentFace->vertCount = 0;
			//if number wasn't read, represents it unused by leaving the values 0
			vertAttribute[0] = 0;
			vertAttribute[1] = 0;
			vertAttribute[2] = 0;
			//start of proper parsing of current line
			while (inputBuffer[inpBufferCursor] != '\0') {
				//check if cursor is at dividing character
				if (inputBuffer[inpBufferCursor] == '/') {
					indexType++;
					inpBufferCursor++;
				}
				//check if cursor is at a number
				else if (inputBuffer[inpBufferCursor] <= 57 && inputBuffer[inpBufferCursor] >= 48) {
					//manual integer read to buffer, and then read with sscanf for stability
					memset(indexBuffer, 0, sizeof(indexBuffer));
					indexBufferCursor = 0;
					while (inputBuffer[inpBufferCursor] <= 57 && inputBuffer[inpBufferCursor] >= 48) {
						indexBuffer[indexBufferCursor] = inputBuffer[inpBufferCursor];
						indexBufferCursor++;
						inpBufferCursor++;
					}
					if (sscanf(indexBuffer, "%d", &vertAttribute[indexType]) != 1) {
						printf("erroneous data in index buffer while parsing obj file. the file may be damaged or corrupted.\n");
						return 1;
					}
					if (indexType == 0) {
						//if index type is vertex position
						currentFace->geometryCenter = vectorAdd(currentFace->geometryCenter, inpScene->vertData[vertAttribute[indexType] - 1]);
					}
					else if (indexType == 2) {
						//if index type is vertex normal
						currentFace->faceNormal = vectorAdd(currentFace->faceNormal, inpScene->vertNormalData[vertAttribute[indexType] - 1]);
					}
				}
				//if cursor points to neither a number nor a dividing char
				//endline concludes face data read
				else {
					currentFace->vertCount += 1;
					//wavefront faces are in order of location - texture - normals
					addVertAttribute(&currentFace->point, vertAttribute[0], inpScene->vertCount, currentFace->vertCount);
					addVertAttribute(&currentFace->texture, vertAttribute[1], inpScene->vertCount, currentFace->vertCount);
					addVertAttribute(&currentFace->normal, vertAttribute[2], inpScene->vertCount, currentFace->vertCount);
					indexType = 0;
					if ((inputBuffer[inpBufferCursor] > 57 || inputBuffer[inpBufferCursor] < 48) && inputBuffer[inpBufferCursor] != ' ' && inputBuffer[inpBufferCursor] == '/') {
						//if the read text isn't a number nor a space nor a '/'
						printf("erroneous data in index buffer while parsing obj file. the file may be damaged or corrupted.\n");
						return 1;
					}
					while (inputBuffer[inpBufferCursor] == ' ')
						inpBufferCursor++;
					if (inputBuffer[inpBufferCursor] == '\n') {
						currentFace->geometryCenter = vectorScale(1 / (double)(currentFace->vertCount), currentFace->geometryCenter);
						currentFace->faceNormal = vectorScale(1 / (double)(currentFace->vertCount), currentFace->faceNormal);
						break;
					}
				}
			}
			break;
		case 'o':
			//object
			inpScene->objectCount++;
			heapMemPtr = realloc(inpScene->objects, sizeof(object) * inpScene->objectCount);
			if (heapMemPtr == NULL) {
				printf("error while allocating memory for sceneObjects\n");
				return 1;
			}
			inpScene->objects = (object*)heapMemPtr;
			inpBufferCursor = 2;
			currentObj = &(inpScene->objects[inpScene->objectCount - 1]);
			currentObj->faceCount = 0;
			currentObj->shadeSmooth = 0;
			currentObj->name = NULL;
			currentObj->mesh = NULL;
			while (inputBuffer[inpBufferCursor] != '\n') {
				heapMemPtr = realloc(currentObj->name, sizeof(char) * (inpBufferCursor - 1));
				if (heapMemPtr == NULL) {
					printf("error while allocating memory for object name parsing.\n");
					return 1;
				}
				currentObj->name = (char*)heapMemPtr;
				currentObj->name[inpBufferCursor - 2] = inputBuffer[inpBufferCursor];
				inpBufferCursor++;
			}
			break;
		case 's':
			//shade smooth
			argCount = sscanf(inputBuffer, "s %d", &castIntToBool);
			currentObj->shadeSmooth = castIntToBool;
			if (argCount != 1) {
				printf("error while reading shade smooth options from provided obj file. the file may be damaged or corrupted.\n");
				return 1;
			}
			break;
		default:
			break;
		}
		fgets(inputBuffer, 256, modelInputOBJ);
	}
	return 0;
}

void initScene(scene* inpScene) {
	inpScene->vertCount = 0;
	inpScene->vertNormalCount = 0;
	inpScene->vertTextureCount = 0;
	inpScene->vertData = NULL;
	inpScene->vertNormalData = NULL;
	inpScene->vertTextureData = NULL;
	inpScene->objectCount = 0;
	inpScene->objects = NULL;
	return;
}

void freeScene(scene* inpScene) {
	free(inpScene->vertData);
	free(inpScene->vertNormalData);
	free(inpScene->vertTextureData);
#pragma warning( push )
#pragma warning( disable : 6001 )
	for (unsigned int tarObj = 0; tarObj < inpScene->objectCount; tarObj++) {
		free(inpScene->objects[tarObj].name);
		for (unsigned int tarFace = 0; tarFace < inpScene->objects[tarObj].faceCount; tarFace++) {
			free(inpScene->objects[tarObj].mesh[tarFace].point);
			free(inpScene->objects[tarObj].mesh[tarFace].texture);
			free(inpScene->objects[tarObj].mesh[tarFace].normal);
		}
		free(inpScene->objects[tarObj].mesh);
	}
#pragma warning( pop )
	return;
}

//FOV is measured on the X axis
//size of projection plane: X = cameraScale, Y = (resolutionY / resolutionX) * cameraScale
void main() {
	//parser variables
	void* heapMemPtr;
	vector vectorBuffer;
	char outputFilename[100];

	scene projScene;
	initScene(&projScene);

	//rendering data
	int widthHalf = width / 2, heightHalf = height / 2;
	double FOV = 2.0943951023931954, cameraScale = 50, perspDist;
	const vector worldOriginLV = { .x = 0, .y = 0, .z = 0 };
	const vector zeroVector = { .x = 0, .y = 0, .z = 0 };
	XY(*screenspaceXY) = NULL;
	RGB(*render)[height] = (RGB(*)[height])calloc(width * height, sizeof(RGB));
	RGB colorPicker = { .r = 255, .b = 255, .g = 255 };
	vector cameraLV = { .x = 0, .y = 0, .z = 0 }, cameraRV = { .x = 0, .y = 0, .z = -1 };
	vector perspLV = { .x = 0, .y = 0, .z = 0 };
	vector camTransformVector;

	double startEXECTIME = 0, objParsingEXECTIME = 0, vectorCalcEXECTIME = 0, endEXECTIME = 0;
	startEXECTIME = (double)(clock()) / CLOCKS_PER_SEC;

	FILE* modelInputOBJ = fopen("resources\\test cube.obj", "rt");
	if (modelInputOBJ == NULL) {
		printf("input .obj file not found\n");
		goto freeMem;
	}
	sprintf(outputFilename, "resources\\render result.ppm");
	FILE* renderResult = fopen(outputFilename, "wt");
	//initialize ppm
	fprintf(renderResult, "P3\n%d %d\n255\n", width, height);
	for (int y = 0; y < height; y++) {
		for (int x = 0; x < width; x++) {
			render[x][y].r = 50;
			render[x][y].b = 50;
			render[x][y].g = 50;
		}
	}
	//obj parsing
	parseObj(modelInputOBJ, &projScene);
	objParsingEXECTIME = (double)(clock()) / CLOCKS_PER_SEC;

	//perspectives ect initialization
	perspDist = cameraScale / (2 * tan(FOV / 2));
	printf("perspDist: %lf units\n", perspDist);
	perspLV.z = perspDist;
	heapMemPtr = realloc(screenspaceXY, sizeof(XY) * projScene.vertCount);
	if (heapMemPtr == NULL) {
		printf("error while allocating memory for screenspaceXY.\n");
		goto freeMem;
	}
	screenspaceXY = (XY*)heapMemPtr;
	camTransformVector = vectorSubtract(worldOriginLV, cameraLV);
	for (unsigned int i = 0; i < projScene.vertCount; i++) {
		vectorBuffer = vectorScale(-1 * perspDist / (projScene.vertData[i].z - perspDist), vectorSubtract(vectorAdd(projScene.vertData[i], camTransformVector), perspLV));
		screenspaceXY[i].x = (unsigned int)(vectorBuffer.x * 100 + widthHalf);
		screenspaceXY[i].y = (unsigned int)(vectorBuffer.y * 100 + heightHalf);
	}
	
	//rendering

	for (unsigned int tarObj = 0; tarObj < projScene.objectCount; tarObj++) {
		for (unsigned int tarFace = 0; tarFace < projScene.objects[tarObj].faceCount; tarFace++) {
			renderLine(render, perspDist, projScene.objects[tarObj].mesh[tarFace].geometryCenter, vectorAdd(projScene.objects[tarObj].mesh[tarFace].geometryCenter, projScene.objects[tarObj].mesh[tarFace].faceNormal), camTransformVector, perspLV, colorPicker);
			//renderPoint(render, perspDist, projScene.objects[tarObj].mesh[tarFace].geometryCenter, camTransformVector, perspLV, colorPicker);
			if (vectorDot(projScene.objects[tarObj].mesh[tarFace].faceNormal, cameraRV) <= 0) {
				drawFaceOutline(render, projScene.objects[tarObj].mesh[tarFace], screenspaceXY, colorPicker);
				drawTrigBland(render, screenspaceXY[projScene.objects[tarObj].mesh[tarFace].point[0] - 1], screenspaceXY[projScene.objects[tarObj].mesh[tarFace].point[1] - 1], screenspaceXY[projScene.objects[tarObj].mesh[tarFace].point[2] - 1], colorPicker);
			}
		}
	}

	
	XY trigTest[3];
	trigTest[0].x = 50;
	trigTest[0].y = 50;

	trigTest[1].x = 10;
	trigTest[1].y = 10;

	trigTest[2].x = 100;
	trigTest[2].y = 20;

	drawTrigBland(render, trigTest[0], trigTest[1], trigTest[2], colorPicker);
	
	vectorCalcEXECTIME = (double)(clock()) / CLOCKS_PER_SEC;

	//writing whats in array 'render' to the actual file
	for (int y = height - 1; y >= 0; y--) {
		for (int x = 0; x < width; x++) {
			fprintfRGB(renderResult, render[x][y].r, render[x][y].g, render[x][y].b);
		}
		fputc('\n', renderResult);
	}

freeMem:
	fclose(modelInputOBJ);
	fclose(renderResult);
	free(render);
	free(screenspaceXY);
	freeScene(&projScene);
	free(projScene.objects);
	endEXECTIME = (double)(clock()) / CLOCKS_PER_SEC;

	printf(".obj file parsing: %lf\nvector calculation: %lf\nwritting the result to file: %lf\nTOTAL TIME: %lf\n", objParsingEXECTIME - startEXECTIME, vectorCalcEXECTIME - objParsingEXECTIME, endEXECTIME - vectorCalcEXECTIME, endEXECTIME - startEXECTIME);
	return;
}
